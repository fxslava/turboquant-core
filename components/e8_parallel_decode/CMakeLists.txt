cmake_minimum_required(VERSION 3.22)
project(e8_parallel_decode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
enable_testing()
foreach(name IN ITEMS test_parallel test_api)
  add_executable(${name} tests/${name}.cpp)
  target_include_directories(${name} PRIVATE src)
  target_include_directories(${name} SYSTEM PRIVATE src/frozen)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
  add_test(NAME ${name} COMMAND ${name})
endforeach()
add_executable(reader_fixture src/reader_runner.cpp)
target_compile_definitions(reader_fixture PRIVATE E8_READER_HOST_ONLY=1)
target_include_directories(reader_fixture SYSTEM PRIVATE src/frozen)
target_compile_options(reader_fixture PRIVATE -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off)
add_test(NAME reader_fixture_uniform COMMAND reader_fixture --golden-root ${CMAKE_CURRENT_SOURCE_DIR}/goldens)
add_test(NAME reader_fixture_nonuniform COMMAND reader_fixture --nonuniform --golden-root ${CMAKE_CURRENT_SOURCE_DIR}/goldens)

option(E8_PARALLEL_BUILD_DEVICE "Compile optional CANN device libraries" OFF)
if(E8_PARALLEL_BUILD_DEVICE)
  set(RUN_MODE sim CACHE STRING "sim or npu")
  include(src/cmake/CannToolchain.cmake)
  include(src/cmake/CannWrapperWarnings.cmake)
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  ascendc_library(e8_parallel_dequant SHARED src/parallel_component.cpp)
  ascendc_library(e8_fused_decode SHARED src/e8_fused_decode.cpp)
  foreach(name IN ITEMS e8_parallel_dequant e8_fused_decode)
    ascendc_compile_definitions(${name} PRIVATE ASCEND_MICRO_CANN_LAUNCH ENABLE_CV_COMM_VIA_SSBUF=1)
    ascendc_compile_options(${name} PRIVATE
      -std=c++17 -Wall -Wextra -Werror -fno-fast-math -ffp-contract=off ${TQ_CANN_SYSTEM_OPTIONS}
      -forward-options-to-host-compiler -Wall -Wextra -Werror ${TQ_CANN_SYSTEM_OPTIONS}
      -DASCEND_MICRO_CANN_LAUNCH -DENABLE_CV_COMM_VIA_SSBUF=1)
  endforeach()
  tq_scope_cann_wrapper_warnings(e8_parallel_dequant e8_fused_decode)
endif()
